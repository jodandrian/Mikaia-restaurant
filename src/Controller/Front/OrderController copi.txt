<?php

namespace App\Controller\Admin;

use App\Entity\Order; // Ou Commande selon votre entité
use App\Repository\OrderRepository;
use Doctrine\ORM\EntityManagerInterface;
use Symfony\Bundle\FrameworkBundle\Controller\AbstractController;
use Symfony\Component\HttpFoundation\Request;
use Symfony\Component\HttpFoundation\Response;
use Symfony\Component\Routing\Attribute\Route;

#[Route('/admin/order')]
class OrderController extends AbstractController
{
    #[Route('/', name: 'app_admin_order_index', methods: ['GET'])]
    public function index(OrderRepository $orderRepository): Response
    {
        return $this->render('admin/order/index.html.twig', [
            'orders' => $orderRepository->findBy([], ['createdAt' => 'DESC']),
        ]);
    }

    #[Route('/{id}/status', name: 'app_admin_order_status', methods: ['POST'])]
    public function updateStatus(Order $order, Request $request, EntityManagerInterface $em): Response
    {
        $status = $request->request->get('status');
        if ($status && $this->isCsrfTokenValid('status'.$order->getId(), $request->request->get('_token'))) {
            $order->setStatus($status);
            $em->flush();
            $this->addFlash('success', 'Statut de la commande mis à jour !');
        }

        return $this->redirectToRoute('app_admin_order_index');
    }
}
